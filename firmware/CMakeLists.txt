cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  set(NEUROGENIC_PICO_DEFAULT ON)
else()
  set(NEUROGENIC_PICO_DEFAULT OFF)
endif()
option(NEUROGENIC_PICO "Build RP2040 firmware (requires PICO_SDK_PATH)" ${NEUROGENIC_PICO_DEFAULT})
if(NEUROGENIC_PICO)
  include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(neurogenic_controller_firmware VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/..
  OUTPUT_VARIABLE NG_SOURCE_GIT_SHA OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
string(LENGTH "${NG_SOURCE_GIT_SHA}" NG_SOURCE_GIT_SHA_LENGTH)
if(NOT NG_SOURCE_GIT_SHA_LENGTH EQUAL 40 OR NG_SOURCE_GIT_SHA MATCHES "[^0-9a-f]")
  set(NG_SOURCE_GIT_SHA "unknown")
endif()
execute_process(COMMAND git status --porcelain --untracked-files=normal --ignore-submodules=dirty
  WORKING_DIRECTORY ${CMAKE_CURRENT_LIST_DIR}/.. OUTPUT_VARIABLE NG_GIT_STATUS
  OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
if(NG_GIT_STATUS STREQUAL "")
  set(NG_SOURCE_DIRTY 0)
else()
  set(NG_SOURCE_DIRTY 1)
endif()
include(${CMAKE_CURRENT_LIST_DIR}/../cmake/GenerateManifest.cmake)
add_library(neurogenic_core src/core.cpp src/illumination.cpp src/rendering.cpp)
target_include_directories(neurogenic_core PUBLIC include generated)
target_compile_definitions(neurogenic_core PUBLIC NG_FIRMWARE_VERSION="0.4.0" NG_PROTOCOL_MAJOR=1 NG_PROTOCOL_MINOR=3)
if(NEUROGENIC_PICO)
  pico_sdk_init()
  add_executable(neurogenic_controller src/main.cpp src/pico_gpio.cpp src/pico_illumination.cpp)
  pico_generate_pio_header(neurogenic_controller ${CMAKE_CURRENT_LIST_DIR}/src/sk6812.pio)
  target_include_directories(neurogenic_controller PRIVATE src)
  target_link_libraries(neurogenic_controller neurogenic_core pico_stdlib hardware_gpio hardware_timer hardware_pio hardware_dma hardware_clocks)
  target_link_options(neurogenic_controller PRIVATE -Wl,-Map=${CMAKE_CURRENT_BINARY_DIR}/neurogenic_controller.map)
  pico_enable_stdio_usb(neurogenic_controller 1)
  pico_enable_stdio_uart(neurogenic_controller 0)
  target_compile_definitions(neurogenic_controller PRIVATE PICO_STDIO_USB_STDOUT_TIMEOUT_US=0
    NG_SOURCE_GIT_SHA="${NG_SOURCE_GIT_SHA}" NG_SOURCE_DIRTY=${NG_SOURCE_DIRTY}
    NG_PICO_SDK_VERSION="${PICO_SDK_VERSION_STRING}" NG_DIAGNOSTIC_VERSION=1)
  pico_add_extra_outputs(neurogenic_controller)
endif()
