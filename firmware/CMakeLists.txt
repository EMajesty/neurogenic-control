cmake_minimum_required(VERSION 3.20)
option(NEUROGENIC_PICO "Build RP2040 firmware (requires PICO_SDK_PATH)" OFF)
if(NEUROGENIC_PICO)
  include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
endif()
project(neurogenic_controller_firmware VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
add_library(neurogenic_core src/core.cpp)
target_include_directories(neurogenic_core PUBLIC include generated)
target_compile_definitions(neurogenic_core PUBLIC NG_FIRMWARE_VERSION="0.1.0" NG_PROTOCOL_MAJOR=1 NG_PROTOCOL_MINOR=3)
if(NEUROGENIC_PICO)
  pico_sdk_init()
  add_executable(neurogenic_controller src/main.cpp)
  target_link_libraries(neurogenic_controller neurogenic_core pico_stdlib tinyusb_device tinyusb_board)
  pico_enable_stdio_usb(neurogenic_controller 1)
  pico_enable_stdio_uart(neurogenic_controller 0)
  pico_add_extra_outputs(neurogenic_controller)
endif()
